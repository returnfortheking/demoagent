```python
import json
from langchain.chains import LLMChain
from langchain.prompts import PromptTemplate
from src.config import get_llm

_PROMPT_TEMPLATE = """你是 HiSpark Studio 的智能助手，负责判断用户意图。

可用的 VS Code 命令列表：
- hispark-studio.build: 编译项目
- hispark-studio.rebuild: 重新编译项目
- hispark-studio.clean: 清理项目
- hispark-studio.flash: 烧录固件（危险操作，需确认）
- hispark-studio.stackAnalysis: 栈分析
- hispark-studio.imageAnalysis: 镜像分析
- stopBuild: 停止编译
- portionOfBurn: 部分烧录（危险操作，需确认）

危险命令（requires_confirmation=true）：hispark-studio.flash, portionOfBurn

判断用户消息属于以下哪类：
1. "action" - 用户想执行某个操作（编译/烧录/分析等）
2. "answer" - 用户有疑问或想了解信息

严格按以下 JSON 格式回复，不要有任何其他内容：
- 若为 action: {{"type": "action", "command": "<命令ID>", "requires_confirmation": <true/false>, "description": "<简短描述>"}}
- 若为 answer: {{"type": "answer"}}

用户消息: {message}
JSON:"""

_PROMPT = PromptTemplate(input_variables=["message"], template=_PROMPT_TEMPLATE)
_chain = LLMChain(llm=get_llm(), prompt=_PROMPT)


def classify_intent(message: str) -> dict:
    """
    分类用户意图。
    Returns:
        {"type": "action", "command": str, "requires_confirmation": bool, "description": str}
        或 {"type": "answer"}
    Raises:
        ValueError: 当 LLM 返回无法解析的内容时
    """
    raw = _chain.run(message=message).strip()
    try:
        return json.loads(raw)
    except json.JSONDecodeError as e:
        raise ValueError(f"Invalid JSON from LLM: {raw!r}") from e
```

**Step 3: 实现 `backend/src/chains/qa_chain.py`**

```python
from langchain.chains import RetrievalQA
from langchain_community.vectorstores import Chroma
from langchain_openai import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.schema import Document
from src.config import get_llm
import os


def _get_embeddings():
    return OpenAIEmbeddings(
        model="embedding-3",
        openai_api_key=os.getenv("ZHIPU_API_KEY"),
        openai_api_base="https://open.bigmodel.cn/api/paas/v4/",
    )


def _load_fixtures() -> list[str]:
    """加载 fixtures 目录下的示例文档"""
    fixtures_path = os.path.join(os.path.dirname(__file__), "../../tests/fixtures/sample_docs.md")
    fixtures_path = os.path.normpath(fixtures_path)
    if os.path.exists(fixtures_path):
        with open(fixtures_path, encoding="utf-8") as f:
            return [f.read()]
    return ["HiSpark Studio 是面向海思芯片的嵌入式开发 IDE。"]


def build_retriever(docs: list[str]):
    """从字符串列表构建向量检索器（Chroma 内存模式）"""
    splitter = RecursiveCharacterTextSplitter(chunk_size=500, chunk_overlap=50)
    documents = []
    for text in docs:
        chunks = splitter.split_text(text)
        documents.extend([Document(page_content=c) for c in chunks])
    vectorstore = Chroma.from_documents(documents, embedding=_get_embeddings())
    return vectorstore.as_retriever(search_kwargs={"k": 3})


# 模块级别：启动时加载 fixtures 初始化检索器
_retriever = build_retriever(_load_fixtures())
_qa_chain = RetrievalQA.from_chain_type(
    llm=get_llm(),
    chain_type="stuff",
    retriever=_retriever,
)


def answer_question(question: str) -> str:
    """基于向量库回答问题"""
    return _qa_chain.run(question)
```

```python
from pydantic import BaseModel
from typing import Optional


class ChatRequest(BaseModel):
    message: str
    thread_id: str


class ChatResponse(BaseModel):
    type: str                              # "action" | "answer"
    # action fields
    command: Optional[str] = None
    args: Optional[dict] = None
    requires_confirmation: Optional[bool] = None
    description: Optional[str] = None
    # answer fields
    answer: Optional[str] = None
    sources: Optional[list[str]] = None
```

**Step 5: 创建 `backend/src/api/main.py`**

```python
from fastapi import FastAPI
from src.api.models import ChatRequest, ChatResponse
from src.chains.intent_classifier import classify_intent
from src.chains.qa_chain import answer_question

app = FastAPI(title="HiSpark AI Agent", version="0.1.0")


@app.get("/health")
def health():
    return {"status": "ok"}


@app.post("/chat", response_model=ChatResponse)
def chat(req: ChatRequest) -> ChatResponse:
    intent = classify_intent(req.message)

    if intent["type"] == "action":
        return ChatResponse(
            type="action",
            command=intent.get("command"),
            args=intent.get("args", {}),
            requires_confirmation=intent.get("requires_confirmation", False),
            description=intent.get("description", ""),
        )
    else:
        answer = answer_question(req.message)
        return ChatResponse(type="answer", answer=answer)
```

```python
import gradio as gr
import httpx

API_URL = "http://localhost:8000"


def chat(message: str, history: list) -> str:
    """调用 FastAPI /chat 端点，格式化返回结果"""
    try:
        resp = httpx.post(
            f"{API_URL}/chat",
            json={"message": message, "thread_id": "gradio-session"},
            timeout=30,
        )
        resp.raise_for_status()
        data = resp.json()

        if data["type"] == "action":
            confirmation = " ⚠️ 需要确认" if data.get("requires_confirmation") else ""
            return f"**[操作命令]**{confirmation}\n命令: `{data['command']}`\n说明: {data.get('description', '')}"
        else:
            return data.get("answer", "（无回答）")
    except Exception as e:
        return f"❌ 错误: {e}"


demo = gr.ChatInterface(
    fn=chat,
    title="HiSpark AI Agent v0.1",
    description="通过自然语言控制 HiSpark Studio，或查询开发文档",
    examples=["帮我编译项目", "烧录固件", "HiSpark 支持哪些芯片？"],
)

if __name__ == "__main__":
    demo.launch(server_port=7860)
```

```typescript
export interface ChatResponse {
    type: 'action' | 'answer';
    // action
    command?: string;
    args?: Record<string, unknown>;
    requires_confirmation?: boolean;
    description?: string;
    // answer
    answer?: string;
    sources?: string[];
}

export class ApiClient {
    constructor(private baseUrl: string) {}

    buildUrl(path: string): string {
        return `${this.baseUrl}${path}`;
    }

    parseResponse(raw: Record<string, unknown>): ChatResponse {
        return raw as ChatResponse;
    }

    async sendMessage(message: string, threadId: string): Promise<ChatResponse> {
        const url = this.buildUrl('/chat');
        const response = await fetch(url, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ message, thread_id: threadId }),
        });
        if (!response.ok) {
            throw new Error(`API error: ${response.status}`);
        }
        const raw = await response.json();
        return this.parseResponse(raw);
    }
}

export const defaultClient = new ApiClient('http://localhost:8000');
```

```typescript
import { ChatResponse } from '../client/ApiClient';

type ExecuteFn = (command: string, ...args: unknown[]) => Promise<unknown>;
type ConfirmFn = (message: string) => Promise<boolean>;

export class CommandExecutor {
    constructor(
        private executeCommand: ExecuteFn,
        private confirmUser: ConfirmFn,
    ) {}

    async handle(response: ChatResponse): Promise<void> {
        if (response.type !== 'action' || !response.command) {
            return;
        }

        if (response.requires_confirmation) {
            const message = `确认执行: ${response.description ?? response.command}？此操作不可撤销。`;
            const confirmed = await this.confirmUser(message);
            if (!confirmed) {
                return;
            }
        }

        await this.executeCommand(response.command);
    }
}
```

```python
def evaluate():

        dataset = json.load(f)

    operation_qa = dataset["operation_qa"]
    results = []
    correct = 0

    for item in operation_qa:
        resp = httpx.post(
            f"{API_URL}/chat",
            json={"message": item["input"], "thread_id": "eval"},
            timeout=30,
        )
        data = resp.json()

        predicted_command = data.get("command", "")
        expected_command = item["expected_command"]
        predicted_type = data.get("type", "")
        expected_type = item["expected_type"]

        match = (predicted_type == expected_type) and (predicted_command == expected_command)
        if match:
            correct += 1

        results.append({
            "id": item["id"],
            "input": item["input"],
            "expected_type": expected_type,
            "expected_command": expected_command,
            "predicted_type": predicted_type,
            "predicted_command": predicted_command,
            "match": match,
        })
        print(f"{'✓' if match else '✗'} {item['id']}: {item['input']!r} → {predicted_command!r} (expected: {expected_command!r})")

    total = len(results)
    rate = correct / total if total > 0 else 0

    report = {
        "version": "v0.1",
        "timestamp": datetime.now().isoformat(),
        "total": total,
        "correct": correct,
        "command_match_rate": round(rate, 4),
        "results": results,
    }

    os.makedirs(os.path.dirname(OUTPUT_PATH), exist_ok=True)
    with open(OUTPUT_PATH, "w", encoding="utf-8") as f:
        json.dump(report, f, ensure_ascii=False, indent=2)

    print(f"\n--- 评估结果 ---")
    print(f"命令匹配率: {correct}/{total} = {rate:.1%}")
    print(f"报告已保存至: {OUTPUT_PATH}")


if __name__ == "__main__":
    evaluate()
```
